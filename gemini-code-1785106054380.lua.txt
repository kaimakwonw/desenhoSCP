local dragging, dragInput, dragStart, startPos

local function UpdateDrag(input)
    local delta = input.Position - dragStart
    MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

Header.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        UpdateDrag(input)
    end
end)

local tabCount = 0

local function CreateTab(name, iconAssetId)
    tabCount = tabCount + 1
    local tabIndex = tabCount
    
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.9, 0, 0, 36)
    btn.Position = UDim2.new(0.05, 0, 0, 10 + (tabIndex - 1) * 42)
    btn.BackgroundColor3 = (tabIndex == 1) and C_CARD or C_BG
    btn.Text = ""
    btn.Parent = Sidebar
    
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = btn

    local icon = Instance.new("ImageLabel")
    icon.Size = UDim2.fromOffset(18, 18)
    icon.Position = UDim2.new(0, 10, 0.5, -9)
    icon.BackgroundTransparency = 1
    icon.Image = iconAssetId
    icon.ImageColor3 = (tabIndex == 1) and C_PURPLE or C_TEXT_DIM
    icon.Parent = btn

    local btnText = Instance.new("TextLabel")
    btnText.Size = UDim2.new(1, -38, 1, 0)
    btnText.Position = UDim2.new(0, 34, 0, 0)
    btnText.BackgroundTransparency = 1
    btnText.Text = name
    btnText.Font = Enum.Font.GothamSemibold
    btnText.TextSize = 13
    btnText.TextColor3 = (tabIndex == 1) and C_PURPLE or C_TEXT_DIM
    btnText.TextXAlignment = Enum.TextXAlignment.Left
    btnText.Parent = btn

    local contentFrame = Instance.new("ScrollingFrame")
    contentFrame.Size = UDim2.new(1, -20, 1, -20)
    contentFrame.Position = UDim2.new(0, 10, 0, 10)
    contentFrame.BackgroundTransparency = 1
    contentFrame.ScrollBarThickness = 3
    contentFrame.ScrollBarImageColor3 = C_PURPLE
    contentFrame.Visible = (tabIndex == 1)
    contentFrame.Parent = TabContainer

    local listLayout = Instance.new("UIListLayout")
    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.Padding = UDim.new(0, 10)
    listLayout.Parent = contentFrame

    Tabs[name] = contentFrame
    TabButtons[name] = {Btn = btn, Icon = icon, Text = btnText}

    btn.MouseButton1Click:Connect(function()
        for tabName, data in pairs(TabButtons) do
            local isCurrent = (tabName == name)
            Tabs[tabName].Visible = isCurrent
            
            TweenService:Create(data.Btn, TweenInfo.new(0.2), {
                BackgroundColor3 = isCurrent and C_CARD or C_BG
            }):Play()
            
            TweenService:Create(data.Text, TweenInfo.new(0.2), {
                TextColor3 = isCurrent and C_PURPLE or C_TEXT_DIM
            }):Play()
            
            TweenService:Create(data.Icon, TweenInfo.new(0.2), {
                ImageColor3 = isCurrent and C_PURPLE or C_TEXT_DIM
            }):Play()
        end
    end)

    return contentFrame
end

local function AddSlider(parentTab, titleText, min, max, defaultValue, callback)
    local card = Instance.new("Frame")
    card.Size = UDim2.new(1, 0, 0, 55)
    card.BackgroundColor3 = C_CARD
    card.Parent = parentTab

    local cardCorner = Instance.new("UICorner")
    cardCorner.CornerRadius = UDim.new(0, 8)
    cardCorner.Parent = card

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6, 0, 0, 25)
    label.Position = UDim2.new(0, 12, 0, 4)
    label.BackgroundTransparency = 1
    label.Text = titleText
    label.Font = Enum.Font.GothamMedium
    label.TextSize = 13
    label.TextColor3 = C_TEXT
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = card

    local valLabel = Instance.new("TextLabel")
    valLabel.Size = UDim2.new(0.3, 0, 0, 25)
    valLabel.Position = UDim2.new(0.7, -12, 0, 4)
    valLabel.BackgroundTransparency = 1
    valLabel.Text = tostring(defaultValue)
    valLabel.Font = Enum.Font.GothamBold
    valLabel.TextSize = 13
    valLabel.TextColor3 = C_PURPLE
    valLabel.TextXAlignment = Enum.TextXAlignment.Right
    valLabel.Parent = card

    local barBg = Instance.new("TextButton")
    barBg.Size = UDim2.new(1, -24, 0, 8)
    barBg.Position = UDim2.new(0, 12, 0, 35)
    barBg.BackgroundColor3 = Color3.fromRGB(40, 40, 55)
    barBg.Text = ""
    barBg.Parent = card

    local barBgCorner = Instance.new("UICorner")
    barBgCorner.CornerRadius = UDim.new(1, 0)
    barBgCorner.Parent = barBg

    local fill = Instance.new("Frame")
    local startPct = math.clamp((defaultValue - min) / (max - min), 0, 1)
    fill.Size = UDim2.new(startPct, 0, 1, 0)
    fill.BackgroundColor3 = C_PURPLE
    fill.BorderSizePixel = 0
    fill.Parent = barBg

    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(1, 0)
    fillCorner.Parent = fill

    local sliding = false
    local function UpdateSlider(input)
        local pos = math.clamp((input.Position.X - barBg.AbsolutePosition.X) / barBg.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * pos)
        fill.Size = UDim2.new(pos, 0, 1, 0)
        valLabel.Text = tostring(val)
        pcall(callback, val)
    end

    barBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = true
            UpdateSlider(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            UpdateSlider(input)
        end
    end)
end

local function ValidarEquipeAlvo(player)
    if not player or player == LocalPlayer then return false end
    if Config.TeamCheck and player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then
        return false
    end

    if player.Team then
        local nomeTime = string.lower(player.Team.Name)
        if string.find(nomeTime, "police") or string.find(nomeTime, "cop") or string.find(nomeTime, "guard") then
            return Config.TargetPolice
        elseif string.find(nomeTime, "inmate") or string.find(nomeTime, "prisoner") then
            return Config.TargetInmates
        elseif string.find(nomeTime, "criminal") or string.find(nomeTime, "crim") then
            return Config.TargetCriminals
        end
    end
    return true
end

local ESPCache = {}

local function CriarObjetosESP(player)
    if not hasDrawing or ESPCache[player] then return end
    
    pcall(function()
        local box = Drawing.new("Square")
        box.Visible = false
        box.Thickness = 1.5
        box.Filled = false

        local textName = Drawing.new("Text")
        textName.Visible = false
        textName.Size = 13
        textName.Center = true
        textName.Outline = true

        local textDist = Drawing.new("Text")
        textDist.Visible = false
        textDist.Size = 12
        textDist.Center = true
        textDist.Outline = true

        local healthOutline = Drawing.new("Square")
        healthOutline.Visible = false
        healthOutline.Color = Color3.fromRGB(0, 0, 0)
        healthOutline.Thickness = 1
        healthOutline.Filled = true

        local healthFill = Drawing.new("Square")
        healthFill.Visible = false
        healthFill.Thickness = 1
        healthFill.Filled = true

        ESPCache[player] = {
            Box = box,
            Name = textName,
            Distance = textDist,
            HealthBg = healthOutline,
            HealthFg = healthFill
        }
    end)
end

local function DesativarESP(esp)
    if not esp then return end
    pcall(function()
        if esp.Box then esp.Box.Visible = false end
        if esp.Name then esp.Name.Visible = false end
        if esp.Distance then esp.Distance.Visible = false end
        if esp.HealthBg then esp.HealthBg.Visible = false end
        if esp.HealthFg then esp.HealthFg.Visible = false end
    end)
end

local function RemoverESP(player)
    if ESPCache[player] then
        for _, obj in pairs(ESPCache[player]) do
            pcall(function() obj:Remove() end)
        end
        ESPCache[player] = nil
    end
end

Players.PlayerRemoving:Connect(RemoverESP)

local function ObterAlvoMaisProximo()
    local alvo = nil
    local menorDistancia = Config.FOVRadius
    local mousePos = UserInputService:GetMouseLocation()

    for _, p in ipairs(Players:GetPlayers()) do
        if ValidarEquipeAlvo(p) and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local parte = p.Character:FindFirstChild(Config.TargetPart)

            if hum and hum.Health > 0 and parte then
                local screenPos, vis = Camera:WorldToViewportPoint(parte.Position)
                if vis then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    if dist < menorDistancia then
                        menorDistancia = dist
                        alvo = parte
                    end
                end
            end
        end
    end
    return alvo
end

local lastUpdate = 0

RunService.RenderStepped:Connect(function()
    pcall(function()
        if FOVCircle then
            FOVCircle.Position = UserInputService:GetMouseLocation()
            FOVCircle.Radius = Config.FOVRadius
            FOVCircle.Visible = Config.ShowFOV and Config.SilentAim
        end
    end)

    local now = os.clock()
    if now - lastUpdate < Config.UpdateRate then return end
    lastUpdate = now

    pcall(function()
        alvoAtual = Config.SilentAim and ObterAlvoMaisProximo() or nil
    end)

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            CriarObjetosESP(p)
            local esp = ESPCache[p]
            local char = p.Character
            local eval = ValidarEquipeAlvo(p)

            if char and eval then
                pcall(function()
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    local head = char:FindFirstChild("Head")

                    -- OTIMIZAÇÃO CRÍTICA DE HITBOX: Atualiza apenas se o tamanho mudou
                    if hrp and hum and hum.Health > 0 then
                        if Config.HitboxEnabled then
                            if not originalSizes[hrp] then originalSizes[hrp] = hrp.Size end
                            local targetVector = Vector3.new(Config.HitboxSize, Config.HitboxSize, Config.HitboxSize)
                            if hrp.Size ~= targetVector then
                                hrp.Size = targetVector
                                hrp.Transparency = Config.HitboxTransparency
                                hrp.Color = Config.HitboxColor
                            end
                        elseif originalSizes[hrp] then
                            hrp.Size = originalSizes[hrp]
                            hrp.Transparency = 1
                            originalSizes[hrp] = nil
                        end
                    end

                    -- Highlight Otimizado
                    local hl = char:FindFirstChild("ESPHighlightSmooth")
                    if Config.ESPHighlight and hum and hum.Health > 0 then
                        if not hl then
                            hl = Instance.new("Highlight")
                            hl.Name = "ESPHighlightSmooth"
                            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                            hl.Parent = char
                        end
                        local teamColor = p.TeamColor and p.TeamColor.Color or Color3.fromRGB(255, 255, 255)
                        hl.FillColor = teamColor
                        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = math.clamp(0.70 + Config.ESPTransparency, 0, 1)
                        hl.OutlineTransparency = math.clamp(0.3 + Config.ESPTransparency, 0, 1)
                        hl.Enabled = true
                    elseif hl then
                        hl.Enabled = false
                    end

                    -- Atualização de Desenho ESP
                    if esp and hrp and head and hum and hum.Health > 0 then
                        local hrpPos, vis = Camera:WorldToViewportPoint(hrp.Position)
                        if vis then
                            local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                            local legPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))

                            local height = math.abs(headPos.Y - legPos.Y)
                            local width = height / 1.8
                            local teamColor = p.TeamColor and p.TeamColor.Color or Color3.fromRGB(255, 255, 255)

                            local corBox = Config.UseTeamColors and teamColor or Config.ESPBoxColor
                            local corTexto = Config.UseTeamColors and teamColor or Config.ESPTextColor
                            local trans = 1 - Config.ESPTransparency

                            if Config.ESPBox then
                                esp.Box.Size = Vector2.new(width, height)
                                esp.Box.Position = Vector2.new(hrpPos.X - width / 2, hrpPos.Y - height / 2)
                                esp.Box.Color = corBox
                                esp.Box.Transparency = trans
                                esp.Box.Visible = true
                            else
                                esp.Box.Visible = false
                            end

                            if Config.ESPName then
                                esp.Name.Text = p.DisplayName
                                esp.Name.Position = Vector2.new(hrpPos.X, hrpPos.Y - height / 2 - 16)
                                esp.Name.Color = corTexto
                                esp.Name.Transparency = trans
                                esp.Name.Visible = true
                            else
                                esp.Name.Visible = false
                            end

                            if Config.ESPDistance then
                                local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                                local dist = myHRP and math.floor((myHRP.Position - hrp.Position).Magnitude) or 0
                                esp.Distance.Text = dist .. "m"
                                esp.Distance.Position = Vector2.new(hrpPos.X, hrpPos.Y + height / 2 + 2)
                                esp.Distance.Color = corTexto
                                esp.Distance.Transparency = trans
                                esp.Distance.Visible = true
                            else
                                esp.Distance.Visible = false
                            end

                            if Config.ESPHealth then
                                local hpPct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                                local barH = height * hpPct
                                
                                esp.HealthBg.Size = Vector2.new(4, height)
                                esp.HealthBg.Position = Vector2.new(hrpPos.X - width / 2 - 6, hrpPos.Y - height / 2)
                                esp.HealthBg.Transparency = trans
                                esp.HealthBg.Visible = true

                                esp.HealthFg.Size = Vector2.new(2, barH)
                                esp.HealthFg.Position = Vector2.new(hrpPos.X - width / 2 - 5, (hrpPos.Y + height / 2) - barH)
                                esp.HealthFg.Color = Color3.fromRGB(255 * (1 - hpPct), 255 * hpPct, 0)
                                esp.HealthFg.Transparency = trans
                                esp.HealthFg.Visible = true
                            else
                                esp.HealthBg.Visible = false
                                esp.HealthFg.Visible = false
                            end
                        else
                            DesativarESP(esp)
                        end
                    else
                        DesativarESP(esp)
                    end
                end)
            else
                DesativarESP(esp)
            end
        end
    end
end)

if hookmetamethod then
    local oldIndex
    oldIndex = hookmetamethod(game, "__index", newcclosure(function(self, key)
        if not checkcaller() and Config.SilentAim and alvoAtual and alvoAtual:IsDescendantOf(Workspace) and self == Mouse then
            if key == "Hit" then
                return alvoAtual.CFrame
            elseif key == "Target" then
                return alvoAtual
            end
        end
        return oldIndex(self, key)
    end))

    local oldNamecall
    oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        local method = getnamecallmethod()
        if not checkcaller() and Config.SilentAim and alvoAtual and alvoAtual:IsDescendantOf(Workspace) then
            if method == "Raycast" then
                local args = {...}
                if typeof(args[1]) == "Vector3" then
                    args[2] = (alvoAtual.Position - args[1]).Unit * 10000
                    return oldNamecall(self, table.unpack(args))
                end
            elseif method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRayWithWhitelist" then
                local args = {...}
                if typeof(args[1]) == "Ray" then
                    args[1] = Ray.new(args[1].Origin, (alvoAtual.Position - args[1].Origin).Unit * 10000)
                    return oldNamecall(self, table.unpack(args))
                end
            end
        end
        return oldNamecall(self, ...)
    end))
end
